#Intervalos de confianza de Nivel educativo

library(tidyverse)

# Calcular resumen estadístico por nivel educativo
resumen_nivel <- Mfinal %>%
  group_by(nivel) %>% # agrupa por categoría
  summarise(
    n = n(),                              # número de observaciones
    media = mean(p26, na.rm = TRUE),  # promedio
    sd = sd(p26, na.rm = TRUE),       # desviación estándar
    se = sd / sqrt(n),                         # error estándar = s / sqrt(n)
    t_crit = qt(0.975, df = n - 1),            # valor t para 95% de confianza
    IC_inf = media - t_crit * se,              # límite inferior
    IC_sup = media + t_crit * se               # límite superior
  )

ggplot(resumen_nivel, aes(x = factor(nivel), y = media)) +
  geom_point(size = 3, color = "black") +  # puntos del promedio
  geom_errorbar(aes(ymin = IC_inf, ymax = IC_sup), width = 0.2, color = "black") +
  labs(
    title = "Promedio del tiempo leído por nivel educativo (IC 95%)",
    x = "Nivel educativo (0 = sin estudios, 9 = doctorado)",
    y = "Tiempo promedio leído"
  ) +
  theme_minimal()
